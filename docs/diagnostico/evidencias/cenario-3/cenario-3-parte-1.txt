  22: class RepositorioExemplarPostgres(RepositorioExemplar):
  23:     def __init__(self, conexao: PgConnection):
  24:         self._conexao = conexao
  25:
  26:     def buscar_disponibilidade(self, livro_id: int) -> dict:
  27:         """Busca a contagem de exemplares totais e disponíveis para um livro."""
  28:         with self._conexao.cursor() as cursor:
  29:             cursor.execute(
  30:                 """
  31:                 SELECT
  32:                     COUNT(*) AS total,
  33:                     COUNT(*) FILTER (WHERE status = 'DISPONIVEL') AS disponiveis
  34:                 FROM exemplares
  35:                 WHERE livro_id = %s;
  36:                 """,
  37:                 (livro_id,)
  38:             )
  39:             row = cursor.fetchone()
  40:
  41:             if row:
  42:                 total = row[0] if isinstance(row, tuple) else row['total']
  43:                 disponiveis = row[1] if isinstance(row, tuple) else row['disponiveis']
  44:             else:
  45:                 total, disponiveis = 0, 0
  46:
  47:             return {
  48:                 "livro_id": livro_id,
  49:                 "total": total,
  50:                 "disponiveis": disponiveis,
  51:                 "em_estoque": disponiveis > 0
  52:             }
  53:
  54:     def buscar_por_id(self, exemplar_id: int) -> Optional[Exemplar]:
  55:         with self._conexao.cursor() as cur:
  56:             cur.execute(
  57:                 "SELECT id, livro_id, status FROM exemplares WHERE id = %s",
  58:                 (exemplar_id,),
  59:             )
  60:             linha = cur.fetchone()
  61:             if linha is None:
  62:                 return None
  63:             return Exemplar(
  64:                 id=linha[0], livro_id=linha[1], status=StatusExemplar(linha[2])
  65:             )
  66:
  67:     def esta_disponivel(self, exemplar_id: int) -> bool:
  68:         with self._conexao.cursor() as cur:
  69:             cur.execute(
  70:                 """
  71:                 SELECT NOT EXISTS (
  72:                     SELECT 1 FROM emprestimos
  73:                     WHERE exemplar_id = %s AND devolvido_em IS NULL

  18:         chave = self._chave_cache(livro_id)

  20:         # 1. Tenta buscar no Cache (Redis)
  21:         dados_cached = self.redis.get(chave)
  22:         if dados_cached:
  23:             return json.loads(dados_cached)
  24:
